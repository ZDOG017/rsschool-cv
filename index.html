Number Expanded Form

56 → "5*10+6"  |  999 → "9*100+9*10+9"  |  10004 → "1*10000+4"

JavaScript 
function simplify(number) {
  if (number === 0) return "";

  const numberAsString = String(number);
  const parts = [];

  for (let index = 0; index < numberAsString.length; index++) {
    const currentDigit = numberAsString[index];

    if (currentDigit === "0") continue;

    const power = numberAsString.length - index - 1;

    if (power === 0) {
      parts.push(currentDigit);
    } else {
      const value = Math.pow(10, power);
      parts.push(currentDigit + "*" + value);
    }
  }

  return parts.join("+");
}
